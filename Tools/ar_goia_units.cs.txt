using System;
using System.Collections.Generic;
using UnityEngine;
namespace CampusRift.AR
{
    public static class ARGestureUnitTests
    {
        static GestureFrame Frame(int time,string label="Open_Palm",float score=.62f,bool hand=true,int age=0)
        {return new GestureFrame{label=label,score=score,timestampMs=1000+time,frameId=1001+time,handPresent=hand,width=512,height=384,screenCoordinates=true,landmarks=hand?GestureSyntheticLandmarks.Create("Open_Palm",new Vector2(.5f,.5f),.3f):null,acquireMs=1000+time,consumeMs=1000+time+age};}
        // Handwritten anatomical fixture in metres; independent of the classifier's thresholds.
        static readonly Vector3[] Open={new Vector3(0,-.06f,0),new Vector3(-.025f,-.04f,0),new Vector3(-.045f,-.02f,0),new Vector3(-.065f,0,0),new Vector3(-.085f,.02f,0),new Vector3(-.035f,0,0),new Vector3(-.04f,.03f,0),new Vector3(-.04f,.05f,0),new Vector3(-.04f,.075f,0),new Vector3(0,.005f,0),new Vector3(0,.04f,0),new Vector3(0,.065f,0),new Vector3(0,.09f,0),new Vector3(.02f,0,0),new Vector3(.025f,.03f,0),new Vector3(.025f,.05f,0),new Vector3(.025f,.075f,0),new Vector3(.035f,-.005f,0),new Vector3(.045f,.02f,0),new Vector3(.045f,.04f,0),new Vector3(.045f,.06f,0)};
        static GestureFrame Fixture(Quaternion rotation,bool mirror=false){var f=Frame(0);f.worldLandmarks=new float[63];for(int i=0;i<21;i++){var p=rotation*Open[i];if(mirror)p.x=-p.x;f.worldLandmarks[i*3]=p.x;f.worldLandmarks[i*3+1]=p.y;f.worldLandmarks[i*3+2]=p.z;}return f;}
        public static object Run()
        {
            var passed=new List<string>();var failed=new List<string>();Action<bool,string> check=(ok,name)=>(ok?passed:failed).Add(name);
            foreach(int hz in new[]{10,12,15,20})foreach(float score in new[]{.6f,.62f,.9f})
            {var m=new GestureStateMachine();var events=new List<GestureIntent>();m.Intent+=events.Add;for(int i=0;i<hz*10;i++)m.Process(Frame((int)Math.Round(i*1000.0/hz),score:score));int expected=hz==10?200:hz==12?167:hz==15?133:100;check(events.Count==1&&events[0].captureSpanMs>=expected-.1&&events[0].captureSpanMs<=expected+.1,$"R2 cadence {hz} score{score}: {expected}ms / one intent");}
            string[] cases={"none_present","held_then_none","single_wrong_label","alternating","reject_then_ready","stable_switch","real_release","stale","unmapped","below_threshold"};int[] expectedCount={0,1,1,0,1,2,2,0,0,0};
            for(int c=0;c<cases.Length;c++){var m=new GestureStateMachine();int count=0;m.Intent+=i=>{count++;if(cases[c]=="reject_then_ready")m.RequireRelease();};for(int i=0;i<200;i++){string label=cases[c]=="none_present"||cases[c]=="held_then_none"&&i>=10&&i<30?"None":cases[c]=="single_wrong_label"&&i==20||cases[c]=="alternating"&&i%2==0||cases[c]=="stable_switch"&&i>=10?"Closed_Fist":cases[c]=="unmapped"?"Thumb_Up":"Open_Palm";m.Process(Frame(i*50,label,cases[c]=="below_threshold"?.59f:.62f,!(cases[c]=="real_release"&&i>=10&&i<=14),cases[c]=="stale"?300:0));}check(count==expectedCount[c],"R2 "+cases[c]);}
            var duplicate=new GestureStateMachine();int dupe=0;duplicate.Intent+=i=>dupe++;foreach(int t in new[]{0,50,50,100})duplicate.Process(Frame(t));check(dupe==1,"R2 duplicate frames ignored");
            var gap=new GestureStateMachine();int gc=0;gap.Intent+=i=>gc++;gap.Process(Frame(0));gap.Process(Frame(1000));check(gc==0,"R2 capture gap resets candidate");
            var absentGap=new GestureStateMachine();int ac=0;absentGap.Intent+=i=>ac++;for(int i=0;i<6;i++)absentGap.Process(Frame(i*50));absentGap.Process(Frame(500,hand:false));absentGap.Process(Frame(1000,hand:false));for(int i=0;i<10;i++)absentGap.Process(Frame(1050+i*50));check(ac==1,"R2 absence gap does not release held");
            var pause=new GestureStateMachine();int pc=0;pause.Intent+=i=>pc++;foreach(int t in new[]{0,50,100})pause.Process(Frame(t));pause.Suspend();for(int t=500;t<=1000;t+=50)pause.Process(Frame(t));check(pc==1,"R2 pause requires real release");for(int t=1050;t<=1250;t+=50)pause.Process(Frame(t,hand:false));foreach(int t in new[]{1300,1350,1400})pause.Process(Frame(t));check(pc==2,"R2 resumes after 200ms / three absence images");
            var jitter=new GestureStateMachine();int jc=0;jitter.Intent+=i=>jc++;foreach(int t in new[]{0,33,100,166,200,266,300,366})jitter.Process(Frame(t));check(jc==1,"R2 jitter one intent");
            var epoch=new GestureStateMachine();int ec=0;epoch.Intent+=i=>ec++;var e1=Frame(0);e1.epoch=2;epoch.Process(e1);e1=Frame(50);e1.epoch=1;epoch.Process(e1);e1=Frame(100);e1.epoch=2;epoch.Process(e1);check(ec==0,"old epoch never adds candidate sample");
            foreach(var rot in new[]{Quaternion.identity,Quaternion.Euler(29,83,41)})foreach(bool reflection in new[]{false,true}){var f=Fixture(rot,reflection);var g=GestureGeometry.Evaluate(f);check(g.quality&&g.label=="Open_Palm","independent 3D fixture rotation/reflection "+rot+reflection);}
            Vector3 a=new Vector3(.9659258f,.258819f,0),b=new Vector3(-.9659258f,.258819f,0);check(Mathf.Abs(GestureGeometry.JointAngle(a,Vector3.zero,b)-150)<.001f,"independent 150 degree angle");
            foreach(int width in new[]{2400,1600}){var f=Fixture(Quaternion.Euler(53,9,70),true);f.width=width;f.height=width*9/20;check(GestureGeometry.Evaluate(f).label=="Open_Palm","aspect does not distort world finger pattern "+width);}
            var invalid=Fixture(Quaternion.identity);invalid.worldLandmarks[0]=float.NaN;check(!GestureGeometry.Evaluate(invalid).quality&&!GestureGeometry.Evaluate(invalid).Contradicts("Victory"),"NaN world is Neutral, never veto");invalid.worldLandmarks=null;check(!GestureGeometry.Evaluate(invalid).Contradicts("Victory"),"missing world is Neutral");invalid.worldLandmarks=new float[63];check(!GestureGeometry.Evaluate(invalid).quality,"degenerate world rejected numerically");
            var edge=Frame(0);edge.landmarks[12]=.999f;check(!GestureGeometry.Evaluate(edge).inFrame,"essential thumb tip clipped rejects framing");
            var thumb=new GestureStateMachine();int tc=0;thumb.Intent+=i=>tc++;for(int t=0;t<1000;t+=50)thumb.Process(Frame(t,"Thumb_Up"));check(tc==0,"ThumbUp remains unmapped, no rescue");
            foreach(int fps in new[]{30,60}){var scheduler=new ARFrameDeadline();int n=0;for(int tick=0;tick<fps*10;tick++)if(scheduler.Due((double)tick/fps,20))n++;check(n>=199&&n<=200,"20Hz deadline at "+fps+" render FPS");}
            var clock=new ARFrameDeadline();check(clock.Unique(4)&&!clock.Unique(4)&&clock.Unique(4.1),"sensor timestamp dedupe");check(clock.Due(0,20)&&clock.Due(1,20)&&!clock.Due(1,20),"missed deadlines produce one attempt, no burst");
            var recovery=new ARRecoveryGate();check(!ARRecoveryGate.Watchdog(999,0,true)&&ARRecoveryGate.Watchdog(1000,0,true),"watchdog 1 second");check(recovery.Begin(1000)&&!recovery.Begin(1001)&&!recovery.Overdue(2999)&&recovery.Overdue(3001),"one recovery in flight and 2 second close deadline");recovery.Complete();check(recovery.Begin(4000),"completed close allows later recovery");
            var polygon=new[]{new Vector2(-1,-1),new Vector2(-1,1),new Vector2(1,1),new Vector2(1,-1)};check(ARPlaneScoring.DiscFits(polygon,Vector2.zero,.5f)&&!ARPlaneScoring.DiscFits(polygon,new Vector2(.8f,0),.5f),"disc center and edges enforce fit");var concave=new[]{new Vector2(-1,-1),new Vector2(-1,1),new Vector2(0,0),new Vector2(1,1),new Vector2(1,-1)};check(!ARPlaneScoring.DiscFits(concave,new Vector2(0,.1f),.18f),"concave gap rejects disc");
            Vector3 point;check(!ARSkillCaster.DiscAim(new Ray(new Vector3(0,1,0),Vector3.right),Vector3.zero,Vector3.up,.5f,out point)&&!ARSkillCaster.DiscAim(new Ray(new Vector3(0,1,0),Vector3.up),Vector3.zero,Vector3.up,.5f,out point)&&!ARSkillCaster.DiscAim(new Ray(new Vector3(1,1,0),Vector3.down),Vector3.zero,Vector3.up,.5f,out point)&&ARSkillCaster.DiscAim(new Ray(new Vector3(0,1,0),Vector3.down),Vector3.zero,Vector3.up,.5f,out point),"aim parallel / behind / outside / valid");
            var hist=new ARMetricHistogram();check(hist.Quantile(.95)==null,"empty quantile null");foreach(double v in new[]{10d,20,30,40,100})hist.Add(v);check(hist.Quantile(.5)==30&&hist.Quantile(.95)==100,"nearest rank quantiles with n");check(ARReviewMetrics.Precision(0,0)==null&&ARReviewMetrics.Precision(3,4)==.75,"precision includes all intents, zero denominator null");
            var metrics=new ARReviewMetrics();for(int i=0;i<30;i++){var row=metrics.cells[i];row.n=4;row.first[row.g]=3;row.timeout=1;row.intents[row.g]=4;row.correctUnique=3;row.multi=1;row.latency.Add(167);}string json=metrics.Export("test","RMX3031","GPU","winner","R","normal-estimated",false,new[]{512,384},"smoke");var parsed=Newtonsoft.Json.Linq.JObject.Parse(json);check(parsed["cells"].Count()==30&&parsed["cells"][0].Count()==24&&System.Text.Encoding.UTF8.GetByteCount("[ARCheck] "+json)<=3500,"30 complete tuples JSON roundtrip <=3500 bytes");
            System.IO.File.WriteAllText("task/ar/goiA/unit-export.json",json);return new{passed,failed};
        }
    }
}
